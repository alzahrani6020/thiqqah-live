rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper: is the user authenticated?
    function isAuth() {
      return request.auth != null;
    }

    // Helper: check role from /roles/{uid}
    function getRole() {
      return get(/databases/$(database)/documents/roles/$(request.auth.uid)).data.role;
    }

    function isAdmin() {
      return isAuth() && getRole() == 'admin';
    }

    function isStaff() {
      return isAuth() && getRole() == 'staff';
    }

    function ownsDoc(resource) {
      return isAuth() && resource.data.senderUid == request.auth.uid;
    }

    // Documents collection
    match /documents/{docId} {
      // Staff/Admin: authenticated users can create documents they own
      allow create: if isAuth()
        && request.resource.data.senderUid == request.auth.uid
        && request.resource.data.status == 'pending';

      // Admin: full read access
      // Staff: read returned docs and their own docs
      // Users: read their own docs
      allow read: if isAdmin()
        || (isStaff() && resource.data.status == 'returned')
        || ownsDoc(resource);

      // Admin: full update (e.g. stamp & return)
      // Sender: can update their own pending doc only
      allow update: if isAdmin()
        || (ownsDoc(resource) && resource.data.status == 'pending' && request.resource.data.status == 'pending');

      // Admin only
      allow delete: if isAdmin();
    }

    // Roles collection (managed manually in Firebase Console)
    match /roles/{uid} {
      allow read: if isAuth() && request.auth.uid == uid;
      allow write: if false; // Only admin via Console
    }

  }
}
